以下 5 种修改状态值的写法都是正确的，毫无疑问加了「⭐️」的是最好的写法

```rust
fn App() -> Element {
    let mut signal = use_signal(|| 0);
    rsx! {
        button {
            onclick: move |_| {
                signal += 1; // 推荐，最简洁 ⭐️
                *signal.write() += 20; // 手动获取可变访问
                signal.set(signal() + 300); // 先读取，再重新设置

                // 测试啰嗦写法
                let mut value = signal.write();
                *value += 4000;

                // 测试文档中的更啰嗦的写法
                // let mut value: &mut i32 = &mut signal.write();
                // *value += 4000;
            },
            "信号值为 {signal}"
        }
    }
}
```
